#include <stdio.h>

int main()
{
    int a, b;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    if (b == 0)
        printf("Division by zero is not allowed");
    else if (a % b == 0)
        printf("Number is divisible");
    else
        printf("Number is not divisible");

    return 0;
}
